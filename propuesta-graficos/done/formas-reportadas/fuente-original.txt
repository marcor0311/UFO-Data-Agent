function categories(rows,o={}){
 const dimension=o.dimension||'shape',all=group(rows,dimension),searched=all.filter(g=>g.key.toLowerCase().includes(String(o.query||'').toLowerCase())),page=Math.max(0,Math.min(+o.page||0,Math.ceil(searched.length/10)-1)),shown=searched.slice(page*10,page*10+10),max=Math.max(1,...all.map(g=>g.n)),h=Math.max(220,shown.length*39+64),x=208,right=688,groups=new Map();let b='';
 [0,.5,1].forEach(t=>{const xx=x+t*(right-x);b+=line(xx,22,xx,h-30,'stroke="'+GRID+'"')+text(xx,h-9,fmt(max*t,0),'text-anchor="middle"')});
 shown.forEach((g,i)=>{groups.set(String(i),g);const y=29+i*39,width=g.n/max*(right-x),label=g.key+'; '+g.n+' de '+rows.length+' reportes, '+fmt(g.n/Math.max(1,rows.length)*100)+'%';b+=pick(text(x-14,y+17,short(g.key,25),'text-anchor="end"')+'<rect class="ua-hit" x="'+x+'" y="'+y+'" width="'+Math.max(1,width)+'" height="24" rx="3" fill="'+(g.key==='Sin clasificar'?'#8895a1':BLUE)+'"/>'+text(711,y+17,fmt(g.n,0),'class="ua-value"')+text(802,y+17,fmt(100*g.n/Math.max(1,rows.length))+'%','text-anchor="end"'),String(i),label)});
 const chart=svg(b,h,'Categorías por frecuencia, conteos y porcentajes del total');
 return {groups,svgs:[chart],html:shell({title:dimension==='shape'?'¿Qué formas se describen en los reportes?':'¿Qué bases aportan esta selección?',lead:'Ranking descriptivo con la misma base de '+fmt(rows.length,0)+' reportes. Las categorías menos frecuentes siguen disponibles.',stats:metric('Categorías',fmt(all.length,0))+metric('Sin clasificar',fmt(rows.filter(r=>r.shape==='Sin clasificar').length,0),'no se sustituyen por texto libre'),controls:select('dimension','Comparar',[['shape','Forma estandarizada'],['source','Base de procedencia']],dimension)+'<label>Localizar categoría<input data-ua-option="query" type="search" value="'+esc(o.query||'')+'" placeholder="Ej.: Triangle"></label>',chart,note:'Cada fila permite examinar los registros. El porcentaje usa el total del alcance, nunca el total de esta página.',extra:'<div class="ua-pages"><button data-ua-page="'+(page-1)+'" '+(page===0?'disabled':'')+'>Anterior</button><span>'+Math.min(searched.length,page*10+1)+'–'+Math.min(searched.length,(page+1)*10)+' de '+searched.length+' categorías</span><button data-ua-page="'+(page+1)+'" '+((page+1)*10>=searched.length?'disabled':'')+'>Siguiente</button></div>',method:'<p>Forma: standardized_shape; ausencia = sin clasificar. No se mezcla con shape libre. Las formas son descripciones de los reportes, no identificaciones del objeto. La procedencia es source_database, no un archivo original verificado. No se agregan las categorías restantes en un “Otros” irreversible.</p>'})};
}
